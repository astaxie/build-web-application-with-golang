# 14.4 Kullanıcı validasyonu

Web uygulamaları geliştirme sürecinde kullanıcı kimlik doğrulama, geliştiricilerin sıkça karşılaştığı bir problemdir. Kullanıcı girişi, kayıt ve çıkış gibi işlemlerin yanı sıra genel kimlik doğrulama üç kısma ayrılabilir:

- HTTP Basic ve HTTP Digest kimlik doğrulama
- Üçüncü Parti Kimlik Doğrulama Entegrasyonu: QQ, mikroblog, watercress, OPENID, Google, GitHub, Facebook ve twitter, vb.
- Genellikle session ve çerez doğrulamasına dayanan özel kullanıcı girişi, kayıt ve çıkış

Beego bu üç şeyin hiçbirini doğal olarak desteklemez, ancak bunları gerçekleştirmek için mevcut üçüncü parti açık kaynak kütüphanelerinden kolayca yararlanabilirsiniz. İlk iki kimlik doğrulama çözümü, sonunda entegre edilmek üzere Beego'nun yol haritasında yer almaktadır.

## HTTP basic ve digest kimlik doğrulama

Hem HTTP basic hem de digest kimlik doğrulama, web uygulamaları tarafından yaygın olarak kullanılan nispeten basit tekniklerdir. Bu iki kimlik doğrulama yöntemini destekleyen zaten birçok açık kaynak üçüncü parti kütüphane bulunmaktadır, örneğin:

	github.com/abbot/go-http-auth

Aşağıdaki kod, bu kütüphaneyi kullanarak bir Beego uygulamasında kimlik doğrulamanın nasıl gerçekleştirileceğini göstermektedir:

	package controllers
	
	import (
		"github.com/abbot/go-http-auth"
		"github.com/astaxie/beego"
	)
	
	func Secret(user, realm string) string {
		if user == "john" {
			// parola "hello"
			return "$1$dlPL2MqE$oQmn16q49SqdmhenQuNgs1"
		}
		return ""
	}
	
	type MainController struct {
		beego.Controller
	}
	
	func (this *MainController) Prepare() {
		a := auth.NewBasicAuthenticator("example.com", Secret)
		if username := a.CheckAuth(this.Ctx.Request); username == "" {
			a.RequireAuth(this.Ctx.ResponseWriter, this.Ctx.Request)
		}
	}
	
	func (this *MainController) Get() {
		this.Data["Username"] = "astaxie"
		this.Data["Email"] = "astaxie@gmail.com"
		this.TplNames = "index.tpl"
	}

Yukarıdaki kod, normal çalışma akışının devam etmesine izin vermeden önce kimlik doğrulamayı gerçekleştirmek için Beego'nun `prepare()` fonksiyonundan yararlanır; görebileceğiniz gibi, HTTP kimlik doğrulamasını gerçekleştirmek çok basittir. Digest kimlik doğrulama da hemen hemen aynı şekilde gerçekleştirilebilir.

## OAuth ve OAuth 2 kimlik doğrulama

OAuth ve OAuth 2, şu anda en popüler iki kimlik doğrulama yöntemidir. Neyse ki, bu tür kimlik doğrulamayı gerçekleştiren, github üzerinde bulunan `go.auth` paketi gibi üçüncü parti kütüphaneler mevcuttur.

	github.com/bradrydzewski/go.auth

Aşağıdaki kod, bu kütüphaneyi kullanarak Beego'da GitHub kimlik bilgilerimizi kullanarak OAuth kimlik doğrulamasının nasıl gerçekleştirileceğini göstermektedir:

1. Önce birkaç route ekleyelim

		beego.RegisterController("/auth/login", &controllers.GithubController{})
		beego.RegisterController("/mainpage", &controllers.PageController{})

2. Sonra `GithubController` giriş sayfasıyla ilgileniyoruz:

		package controllers
			
		import (
			"github.com/astaxie/beego"
			"github.com/bradrydzewski/go.auth"
		)
		
		const (
			githubClientKey = "a0864ea791ce7e7bd0df"
			githubSecretKey = "a0ec09a647a688a64a28f6190b5a0d2705df56ca"
		)
		
		type GithubController struct {
			beego.Controller
		}
		
		func (this *GithubController) Get() {
			// auth parametrelerini ayarla
			auth.Config.CookieSecret = []byte("7H9xiimk2QdTdYI7rDddfJeV")
			auth.Config.LoginSuccessRedirect = "/mainpage"
			auth.Config.CookieSecure = false
		
			githubHandler := auth.Github(githubClientKey, githubSecretKey)
		
			githubHandler.ServeHTTP(this.Ctx.ResponseWriter, this.Ctx.Request)
		}

3. Başarılı giriş sayfası sonrası işlem:

		package controllers
		
		import (
			"github.com/astaxie/beego"
			"github.com/bradrydzewski/go.auth"
			"net/http"
			"net/url"
		)
		
		type PageController struct {
			beego.Controller
		}
		
		func (this *PageController) Get() {
			// auth parametrelerini ayarla
			auth.Config.CookieSecret = []byte("7H9xiimk2QdTdYI7rDddfJeV")
			auth.Config.LoginSuccessRedirect = "/mainpage"
			auth.Config.CookieSecure = false
		
			user, err := auth.GetUserCookie(this.Ctx.Request)
		
			//eğer aktif kullanıcı oturumu yoksa kullanıcıyı yetkilendir
			if err != nil || user.Id() == "" {
				http.Redirect(this.Ctx.ResponseWriter, this.Ctx.Request, auth.Config.LoginRedirect, http.StatusSeeOther)
				return
			}
		
			//aksi halde, kullanıcıyı URL'ye ekle ve devam et
			this.Ctx.Request.URL.User = url.User(user.Id())
			this.Data["pic"] = user.Picture()
			this.Data["id"] = user.Id()
			this.Data["name"] = user.Name()
			this.TplNames = "home.tpl"
		}

## Özel kimlik doğrulama

Özel kimlik doğrulama genellikle session doğrulamasıyla birlikte kullanılır; aşağıdaki kod, bunu gösteren Beego tabanlı açık kaynak bir blogdur:

	//Giriş işlemi
	func (this *LoginController) Post() {
		this.TplNames = "login.tpl"
		this.Ctx.Request.ParseForm()
		username := this.Ctx.Request.Form.Get("username")
		password := this.Ctx.Request.Form.Get("password")
		md5Password := md5.New()
		io.WriteString(md5Password, password)
		buffer := bytes.NewBuffer(nil)
		fmt.Fprintf(buffer, "%x", md5Password.Sum(nil))
		newPass := buffer.String()
	
		now := time.Now().Format("2006-01-02 15:04:05")
	
		userInfo := models.GetUserInfo(username)
		if userInfo.Password == newPass {
			var users models.User
			users.Last_logintime = now
			models.UpdateUserInfo(users)
	
			//Başarılı giriş sonrası oturumu ayarla
			sess := globalSessions.SessionStart(this.Ctx.ResponseWriter, this.Ctx.Request)
			sess.Set("uid", userInfo.Id)
			sess.Set("uname", userInfo.Username)
	
			this.Ctx.Redirect(302, "/")
		}	
	}
	
	//Kayıt işlemi
	func (this *RegController) Post() {
		this.TplNames = "reg.tpl"
		this.Ctx.Request.ParseForm()
		username := this.Ctx.Request.Form.Get("username")
		password := this.Ctx.Request.Form.Get("password")
		usererr := checkUsername(username)
		fmt.Println(usererr)
		if usererr == false {
			this.Data["UsernameErr"] = "Kullanıcı adı hatalı, lütfen tekrar deneyin"
			return
		}
	
		passerr := checkPassword(password)
		if passerr == false {
			this.Data["PasswordErr"] = "Parola hatalı, lütfen tekrar deneyin"
			return
		}
	
		md5Password := md5.New()
		io.WriteString(md5Password, password)
		buffer := bytes.NewBuffer(nil)
		fmt.Fprintf(buffer, "%x", md5Password.Sum(nil))
		newPass := buffer.String()
	
		now := time.Now().Format("2006-01-02 15:04:05")
	
		userInfo := models.GetUserInfo(username)
	
		if userInfo.Username == "" {
			var users models.User
			users.Username = username
			users.Password = newPass
			users.Created = now
			users.Last_logintime = now
			models.AddUser(users)
	
			//Başarılı giriş sonrası oturumu ayarla
			sess := globalSessions.SessionStart(this.Ctx.ResponseWriter, this.Ctx.Request)
			sess.Set("uid", userInfo.Id)
			sess.Set("uname", userInfo.Username)
			this.Ctx.Redirect(302, "/")
		} else {
			this.Data["UsernameErr"] = "Kullanıcı zaten var"
		}
	
	}
	
	func checkPassword(password string) (b bool) {
		if ok, _ := regexp.MatchString("^[a-zA-Z0-9]{4,16}$", password); !ok {
			return false
		}
		return true
	}
	
	func checkUsername(username string) (b bool) {
		if ok, _ := regexp.MatchString("^[a-zA-Z0-9]{4,16}$", username); !ok {
			return false
		}
		return true
	}

Kullanıcı girişi ve kaydını gerçekleştirdikten sonra, kullanıcının giriş yapıp yapmadığını belirlemek için diğer modüller eklenebilir:

	func (this *AddBlogController) Prepare() {
		sess := globalSessions.SessionStart(this.Ctx.ResponseWriter, this.Ctx.Request)
		sess_uid := sess.Get("userid")
		sess_username := sess.Get("username")
		if sess_uid == nil {
			this.Ctx.Redirect(302, "/admin/login")
			return
		}
		this.Data["Username"] = sess_username
	}
